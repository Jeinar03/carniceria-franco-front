class CustomerService { // customer-service.service.ts:11-41
  constructor(http) { this.customer = localStorage.getItem('clienteId'); this.http = http; }
  uploadCustomerImages(images) {
    const url = `${environment.apiUrl}/usuarios/store-images`;
    const payload = { customer_id: this.customer, images: images };
    return this.http.post(url, payload);
  }
  getImagenes() { return this.http.get(`${environment.apiUrl}/usuarios/store-images/${this.customer}/cliente`); }
  deleteImage(imageId) { return this.http.delete(`${environment.apiUrl}/usuarios/borrar/${this.customer}/imagenes/${imageId}`); }
}
