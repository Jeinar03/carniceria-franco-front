function method(src, sig) {
  const i = src.indexOf(sig); if (i < 0) throw new Error('no ' + sig);
  let j = src.indexOf('{', i), d = 0, k = j;
  for (; k < src.length; k++) { if (src[k] === '{') d++; else if (src[k] === '}') { d--; if (!d) break; } }
  return src.slice(j, k + 1);
}
